cmake_minimum_required(VERSION 3.15)
project(CToolWolf LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Windows Win32 (x86)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8 /EHsc)
add_executable(inject-x86 injector/main.cpp)
target_link_libraries(inject-x86 PRIVATE advapi32 user32)
add_library(ctool-wolf-x86 SHARED wolf/main.cpp)
add_executable(wolf-fixture WIN32 tests/fixture.cpp)
add_executable(wolf-database-test tests/database_reader_test.cpp)
target_link_libraries(wolf-fixture PRIVATE user32)
