cmake_minimum_required(VERSION 3.15)
project(CToolWolf LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Windows Win32 (x86)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /utf-8 /EHsc)
add_executable(inject-x86 injector/main.cpp)
target_link_libraries(inject-x86 PRIVATE advapi32 user32)
add_library(ctool-wolf-x86 SHARED wolf/main.cpp)
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde32.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
target_link_libraries(ctool-wolf-x86 PRIVATE minhook winmm user32)
add_custom_command(TARGET ctool-wolf-x86 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/vendor/minhook/LICENSE.txt" "$<TARGET_FILE_DIR:ctool-wolf-x86>/MinHook-LICENSE.txt")
add_executable(wolf-fixture WIN32 tests/fixture.cpp)
add_executable(wolf-database-test tests/database_reader_test.cpp)
add_executable(wolf-runtime-test tests/runtime_test.cpp)
target_link_libraries(wolf-runtime-test PRIVATE minhook winmm user32)
target_link_libraries(wolf-fixture PRIVATE user32)
